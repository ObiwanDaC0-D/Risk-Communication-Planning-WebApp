-- Risk Communication Planner: foundational PostgreSQL schema
-- This schema is intentionally data-model only; no authentication, RLS, migrations,
-- workflow automation, or document-generation behavior is implemented in the UI scaffold.

create extension if not exists pgcrypto;

create type plan_status as enum ('draft', 'in_review', 'approved', 'archived');
create type task_status as enum ('not_started', 'in_progress', 'ready_for_review', 'complete');
create type effort_level as enum ('maintain', 'improve', 'substantially_expand', 'surge');
create type measure_type as enum ('baseline', 'discretionary');
create type viability_result as enum ('viable', 'not_viable', 'pending');
create type review_status as enum ('not_requested', 'pending', 'returned', 'approved');

create table organizations (
  id uuid primary key default gen_random_uuid(), name text not null, created_at timestamptz not null default now()
);
create table users (
  id uuid primary key default gen_random_uuid(), organization_id uuid references organizations(id), display_name text not null, email text unique, created_at timestamptz not null default now()
);
create table plans (
  id uuid primary key default gen_random_uuid(), organization_id uuid not null references organizations(id), title text not null,
  status plan_status not null default 'draft', planning_horizon_start date, planning_horizon_end date,
  plan_owner_id uuid references users(id), approval_authority_id uuid references users(id), version integer not null default 1,
  created_at timestamptz not null default now(), updated_at timestamptz not null default now()
);
create table plan_members (
  plan_id uuid references plans(id) on delete cascade, user_id uuid references users(id), role text not null,
  primary key (plan_id, user_id)
);
create table plan_tasks (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  task_number smallint not null check (task_number between 1 and 14), status task_status not null default 'not_started',
  narrative jsonb not null default '{}'::jsonb, completion_checked boolean not null default false,
  validated_by uuid references users(id), validated_at timestamptz, unique (plan_id, task_number)
);
create table evidence_items (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  title text not null, source_type text, source_reference text, collected_on date, summary text,
  assumptions text, handling_note text, validation_status review_status not null default 'not_requested', created_at timestamptz not null default now()
);
create table task_evidence_links (
  task_id uuid references plan_tasks(id) on delete cascade, evidence_id uuid references evidence_items(id) on delete cascade,
  rationale text, primary key (task_id, evidence_id)
);
create table governance_assignments (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  function_name text not null, user_id uuid references users(id), raci_role text not null check (raci_role in ('responsible','accountable','consulted','informed'))
);
create table planning_logic_entries (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  problem text not null, opportunity text, objective text, constraint_text text, implication_for_measures text, sort_order integer not null default 0
);
create table stakeholders (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  name text not null, stakeholder_type text, interest_or_concern text, needed_action text, current_relationship text,
  communication_need text, preferred_channel text, access_accommodation text, priority_rank smallint, owner_id uuid references users(id)
);
create table effort_assessments (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null unique references plans(id) on delete cascade,
  recommended_level effort_level, scoring_method text, participants text, assessed_on date, rationale text, sensitivity_check text
);
create table effort_assessment_factors (
  id uuid primary key default gen_random_uuid(), assessment_id uuid not null references effort_assessments(id) on delete cascade,
  category text not null, rating text not null check (rating in ('low','medium','high')), weight numeric(6,2), rationale text, evidence_id uuid references evidence_items(id)
);
create table measures (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  title text not null, measure_type measure_type not null, objective_id uuid references planning_logic_entries(id),
  target_description text, content_summary text, delivery_method text, frequency text, accountable_owner_id uuid references users(id),
  prerequisites text, risks_limitations text, status text not null default 'candidate'
);
create table alternatives (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  name text not null, description text, target_coverage text, expected_benefit text, resource_assumption text, schedule_assumption text,
  dependencies text, residual_risk text, screening_result viability_result not null default 'pending', selected boolean not null default false
);
create table alternative_measures (
  alternative_id uuid references alternatives(id) on delete cascade, measure_id uuid references measures(id) on delete cascade, primary key (alternative_id, measure_id)
);
create table alternative_screening_records (
  id uuid primary key default gen_random_uuid(), alternative_id uuid not null references alternatives(id) on delete cascade,
  criterion text not null, result viability_result not null, evidence_or_rationale text, evidence_id uuid references evidence_items(id)
);
create table alternative_evaluations (
  id uuid primary key default gen_random_uuid(), alternative_id uuid not null references alternatives(id) on delete cascade,
  criterion text not null, score numeric(8,2), weight numeric(8,2), rationale text
);
create table selection_decisions (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null unique references plans(id) on delete cascade,
  selected_alternative_id uuid references alternatives(id), decision_narrative text, tradeoffs text, resource_implications text,
  residual_risks text, approving_authority_id uuid references users(id), decided_at timestamptz
);
create table budget_lines (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  measure_id uuid references measures(id), fiscal_year text, office text, labor_hours numeric(12,2), unit_cost numeric(14,2),
  travel_materials_cost numeric(14,2), contract_support_cost numeric(14,2), total_cost numeric(14,2), currency_code char(3) not null default 'USD'
);
create table action_items (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  measure_id uuid references measures(id), action_text text not null, accountable_owner_id uuid references users(id),
  supporting_units text, due_date date, recurrence text, status text not null default 'not_started', dependencies text, notes text
);
create table key_messages (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  stakeholder_id uuid references stakeholders(id), need_to_know text, needed_action text, message_text text, limitations text,
  coordination_needs text, content_owner_id uuid references users(id)
);
create table performance_measures (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  measure_id uuid references measures(id), measure_kind text not null check (measure_kind in ('performance','effectiveness')),
  objective text, metric text not null, baseline_value text, target_value text, data_source_method text, owner_id uuid references users(id), cadence text
);
create table reviews_and_aars (
  id uuid primary key default gen_random_uuid(), plan_id uuid not null references plans(id) on delete cascade,
  record_type text not null check (record_type in ('implementation_review','annual_review','after_action_review','revision_trigger')),
  event_date date, finding text, corrective_action text, owner_id uuid references users(id), closure_status text not null default 'open', closure_date date
);
create table audit_events (
  id uuid primary key default gen_random_uuid(), plan_id uuid references plans(id) on delete cascade, actor_id uuid references users(id),
  entity_type text not null, entity_id uuid, action text not null, before_state jsonb, after_state jsonb, occurred_at timestamptz not null default now()
);

create index idx_plan_tasks_plan on plan_tasks(plan_id);
create index idx_evidence_plan on evidence_items(plan_id);
create index idx_stakeholders_plan on stakeholders(plan_id);
create index idx_measures_plan on measures(plan_id);
create index idx_actions_plan on action_items(plan_id);
create index idx_audit_plan_time on audit_events(plan_id, occurred_at desc);
